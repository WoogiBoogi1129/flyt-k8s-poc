cmake_minimum_required(VERSION 3.18)
project(flyt_shm LANGUAGES C)
add_subdirectory(../../experiments/shm-queue queue)
add_library(flyt_mapping STATIC src/mapping.c)
target_compile_features(flyt_mapping PUBLIC c_std_11)
target_include_directories(flyt_mapping PUBLIC include)
target_link_libraries(flyt_mapping PUBLIC flyt_shm_queue)
set_target_properties(flyt_mapping PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(FLYT_BUILD_CUDA_RUNTIME ON CACHE BOOL "Worker requires CUDA runtime backend")
add_subdirectory(../../experiments/cuda-dispatch exec)
find_package(CUDAToolkit 12.8 REQUIRED)
add_library(flyt_dispatch STATIC src/dispatch.c)
target_include_directories(flyt_dispatch PUBLIC include)
target_link_libraries(flyt_dispatch PUBLIC flyt_cuda_exec flyt_mapping)
add_executable(flyt-shm-worker src/worker.c)
add_library(flyt_async STATIC src/async.c)
target_link_libraries(flyt_async PUBLIC flyt_dispatch CUDA::cudart CUDA::cuda_driver)
find_library(FLYT_CUDNN cudnn REQUIRED)
add_library(flyt_compat STATIC src/compat.c)
target_link_libraries(flyt_compat PUBLIC flyt_async CUDA::cublas ${FLYT_CUDNN})
target_link_libraries(flyt-shm-worker PRIVATE flyt_dispatch flyt_cuda_runtime flyt_async flyt_compat)
add_library(flyt_guest SHARED src/guest.c src/guest_async.c src/guest_compat.c)
target_include_directories(flyt_guest PRIVATE ${CUDAToolkit_INCLUDE_DIRS} ../../experiments/cuda-dispatch/include)
target_link_libraries(flyt_guest PRIVATE flyt_mapping)
install(TARGETS flyt-shm-worker flyt_guest RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
